라이트·다크 토큰의 실제 값으로 계산한다. 입력 경계는 라벨과 배경을 함께 쓰므로 제외하고, 체크박스·라디오는 `--line-strong`을 쓴다.

export const pairs = [
  ...["--text-1", "--text-2", "--text-3", "--accent-text", "--success-text", "--warning-text", "--danger-text", "--info-text"].map((name) => [name, name, "--bg-canvas", 4.5]),
  ...["--success", "--warning", "--danger", "--info", "--line-strong", "--focus"].map((name) => [name, name, "--bg-canvas", 3]),
  ...["--accent", "--danger-solid"].map((name) => [name, "--on-accent", name, 4.5]),
  ...[1, 2, 3, 4, 5, 6].map((n) => [`차트 ${n}`, `--chart-${n}`, "--bg-canvas", 4.5]),
];

<ContrastTable pairs={pairs} />

키보드 포커스는 2px `--focus` 링으로 보인다. 동작을 색만으로 구분하지 않는다. 상태는 모양·색·라벨을 함께 쓴다. 터치에서도 같은 컨트롤 크기를 사용한다.
